cmake_minimum_required(VERSION 3.15)
project(CommanderDesktop LANGUAGES CXX)
enable_language(RC)
set(CMAKE_CXX_STANDARD 17)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use a Windows MSVC x86 Native Tools prompt")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(STARCRAFT_LAB "E:/StarCraftAI" CACHE PATH "Local SDK and game lab")
set(BWAPI_ROOT "${STARCRAFT_LAB}/dependencies/bwapi" CACHE PATH "BWAPI SDK")
set(BWEM_ROOT "${STARCRAFT_LAB}/dependencies/strategy-libs/bwem/BWEM" CACHE PATH "BWEM source")
set(TRANSPORT_UPSTREAM "${STARCRAFT_LAB}/dependencies/openbw/bwapi/bwapi" CACHE PATH "BWAPI transport utility sources")
set(TRANSPORT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/local-pc-transport/SNP")
add_library(TransportStormStub SHARED "${TRANSPORT_UPSTREAM}/Storm/storm.cpp" "${TRANSPORT_UPSTREAM}/Storm/storm.def")
target_compile_definitions(TransportStormStub PRIVATE WIN32 NOMINMAX STORM_EXPORTS _CRT_SECURE_NO_WARNINGS)
target_include_directories(TransportStormStub PRIVATE "${TRANSPORT_UPSTREAM}/Storm")
set_target_properties(TransportStormStub PROPERTIES OUTPUT_NAME Storm RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/transport" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/transport")
file(GLOB TRANSPORT_CPP "${TRANSPORT_SOURCE}/*.cpp")
file(GLOB TRANSPORT_UTIL "${TRANSPORT_UPSTREAM}/Util/Source/Util/*.cpp")
add_library(CommanderLocalPC SHARED ${TRANSPORT_CPP} ${TRANSPORT_UTIL} "${TRANSPORT_SOURCE}/UI.rc" "${TRANSPORT_SOURCE}/SNPModule.def")
target_compile_definitions(CommanderLocalPC PRIVATE WIN32 NOMINMAX SNP_DIRECTIP_EXPORTS _CRT_SECURE_NO_WARNINGS)
target_include_directories(CommanderLocalPC PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${TRANSPORT_UPSTREAM}" "${TRANSPORT_UPSTREAM}/Util/Source" "${TRANSPORT_UPSTREAM}/Storm")
target_link_libraries(CommanderLocalPC TransportStormStub ws2_32)
add_custom_command(TARGET CommanderLocalPC POST_BUILD COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/package_snp.py" "$<TARGET_FILE:CommanderLocalPC>" "${CMAKE_CURRENT_BINARY_DIR}/CommanderLocalPC.snp" "${TRANSPORT_SOURCE}/caps.mpq")
add_library(WideView SHARED WideView.cpp)
target_compile_definitions(WideView PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET WideView PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
file(GLOB BWEM_SOURCES "${BWEM_ROOT}/src/*.cpp")
add_library(CommanderLayer SHARED CommanderLayer.cpp Candidate.def ${BWEM_SOURCES})
target_compile_definitions(CommanderLayer PRIVATE NOMINMAX WIN32 NDEBUG _CRT_SECURE_NO_WARNINGS)
target_compile_options(CommanderLayer PRIVATE /bigobj /utf-8)
target_include_directories(CommanderLayer PRIVATE "${BWAPI_ROOT}/include" "${BWEM_ROOT}/include")
target_link_libraries(CommanderLayer "${BWAPI_ROOT}/lib/BWAPI.lib" user32 gdi32)
add_library(ReplayControl SHARED ReplayControl.cpp Candidate.def)
target_compile_definitions(ReplayControl PRIVATE NOMINMAX WIN32 NDEBUG)
target_include_directories(ReplayControl PRIVATE "${BWAPI_ROOT}/include")
target_link_libraries(ReplayControl "${BWAPI_ROOT}/lib/BWAPI.lib" user32 dwmapi)
add_executable(WindowRecorder WindowRecorder.cpp)
target_compile_definitions(WindowRecorder PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(WindowRecorder PRIVATE /bigobj)
option(COMMANDER_ENABLE_SCREEN_CAPTURE "Build the legacy recorder with screen capture support (explicit user request only)" OFF)
if(COMMANDER_ENABLE_SCREEN_CAPTURE)
  target_compile_definitions(WindowRecorder PRIVATE COMMANDER_ENABLE_SCREEN_CAPTURE=1)
  target_link_libraries(WindowRecorder d3d11 dxgi windowsapp user32)
endif()
add_library(LegacyStorm SHARED LegacyStorm.cpp LegacyStorm.def)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/StormBase.lib"
  COMMAND "${CMAKE_AR}" /nologo /machine:x86
    "/def:${CMAKE_CURRENT_SOURCE_DIR}/LegacyStormImport.def"
    "/out:${CMAKE_CURRENT_BINARY_DIR}/StormBase.lib"
  DEPENDS LegacyStormImport.def VERBATIM)
add_custom_target(StormBaseImport DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/StormBase.lib")
add_dependencies(LegacyStorm StormBaseImport)
target_link_libraries(LegacyStorm "${CMAKE_CURRENT_BINARY_DIR}/StormBase.lib")
add_library(ResultsWatcher SHARED ResultsWatcher.cpp ResultsWatcher.def)
target_compile_definitions(ResultsWatcher PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
